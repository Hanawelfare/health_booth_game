ฐานข้อมูลพนักงาน (Auto-Lookup)

เมื่อผู้เล่นกรอกรหัสพนักงาน ระบบจะดึง ชื่อ-นามสกุล และ แผนก จากรายการด้านล่างนี้ขึ้นมาอัตโนมัติ

โค้ด Google Apps Script (ดึงรายชื่อ + บันทึกผล + ส่งประวัติเข้าแดชบอร์ด) กดดูโค้ด
// 1. ส่งข้อมูลรายชื่อพนักงาน และ ประวัติการเล่นเข้าเว็บแดชบอร์ด
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var nameSheet = ss.getSheetByName("Name") || ss.getSheets()[0];
  var nameData = nameSheet.getDataRange().getValues();
  var employees = {};
  for (var i = 1; i < nameData.length; i++) {
    var row = nameData[i];
    var empId = String(row[0] || '').trim().replace(/^'/, '');
    if (empId) employees[empId] = { id: empId, name: String(row[1] || '').trim(), dept: String(row[2] || '-').trim() };
  }
  var logSheet = ss.getSheetByName("GameLogs");
  var logs = [];
  if (logSheet) {
    var logData = logSheet.getDataRange().getValues();
    for (var j = 1; j < logData.length; j++) {
      var lRow = logData[j];
      var lEmpId = String(lRow[1] || '').trim().replace(/^'/, '');
      if (lEmpId) {
        var dateVal = lRow[0];
        var dateStr = (dateVal instanceof Date) ? Utilities.formatDate(dateVal, "Asia/Bangkok", "dd/MM/yyyy HH:mm:ss") : String(dateVal || '');
        logs.push({ roundId: j, dateTime: dateStr, timestamp: dateStr.split(' ')[1] || dateStr, empId: lEmpId, name: String(lRow[2] || '').trim(), dept: String(lRow[3] || '-').trim(), score: Number(lRow[4]) || 0 });
      }
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "success", count: Object.keys(employees).length, employees: employees, logs: logs })).setMimeType(ContentService.MimeType.JSON);
}

// 2. รับผลการแข่งจากจอหลัก บันทึกลงชีต GameLogs (ล็อกเลข 0 นำหน้า พร้อมป้องกันบันทึกซ้ำ)
function doPost(e) {
  try {
    var d = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var s = ss.getSheetByName("GameLogs") || ss.getActiveSheet();
    var empId = "'" + String(d.empIdRaw || d.empId || '').replace(/^'/, '');
    var lastRow = s.getLastRow();
    if (lastRow > 1) {
      var lastEmpId = String(s.getRange(lastRow, 2).getValue()).replace(/^'/, '');
      var lastDate = new Date(s.getRange(lastRow, 1).getValue());
      var now = new Date();
      if (lastEmpId === String(d.empIdRaw || d.empId || '').replace(/^'/, '') && ((now.getTime() - lastDate.getTime()) / 1000) < 3) {
        return ContentService.createTextOutput(JSON.stringify({status:'ignored_duplicate'})).setMimeType(ContentService.MimeType.JSON);
      }
    }
    s.appendRow([new Date(), empId, d.name, d.dept, d.score]);
    return ContentService.createTextOutput(JSON.stringify({status:'success'})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status:'error', message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

    var empId = "'" + String(d.empIdRaw || d.empId || '').replace(/^'/, '');
    s.appendRow([new Date(), empId, d.name, d.dept, d.score]);
    return ContentService.createTextOutput(JSON.stringify({status:'success'})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status:'error', message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
